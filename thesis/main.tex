\newcommand{\DiffCommentIcon}{%
    \tikz[baseline=-0.55ex, x=1ex, y=1ex]{%
        \draw[diffcommentframe, line width=0.35pt, rounded corners=0.25ex] (0,0) rectangle (1.55,1.05);
        \draw[diffcommentframe, line width=0.35pt] (0.42,0) -- (0.22,-0.38) -- (0.75,0);
    }%
}

\long\def\DiffComment#1{%
    \ifdiffversion
        \begin{diffcommentbox}\DiffCommentIcon\hspace{0.5em}\textbf{Komentar mentora:} #1\end{diffcommentbox}%
    \fi
}

\long\def\DiffReplace#1#2{%
    \ifdiffversion
        \begin{diffdeletedbox}#1\end{diffdeletedbox}%
        \begin{diffaddedbox}#2\end{diffaddedbox}%
    \else
        #2%
    \fi
}

\long\def\DiffAdd#1{%
    \ifdiffversion
        \begin{diffaddedbox}#1\end{diffaddedbox}%
    \else
        #1%
    \fi
}

\long\def\DiffDelete#1{%
    \ifdiffversion
        \begin{diffdeletedbox}#1\end{diffdeletedbox}%
    \fi
}

\protected\def\DiffReplaceInline#1#2{%
    \ifdiffversion
        \colorbox{diffdeleted}{\strut #1}\ \colorbox{diffadded}{\strut #2}%
    \else
        #2%
    \fi
}

\protected\def\DiffAddInline#1{%
    \ifdiffversion
        \colorbox{diffadded}{\strut #1}%
    \else
        #1%
    \fi
}

\protected\def\DiffDeleteInline#1{%
    \ifdiffversion
        \colorbox{diffdeleted}{\strut #1}%
    \fi
}
